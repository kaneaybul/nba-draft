resolve_base_dir <- function() {
  candidates <- c(
    getwd(),
    file.path(getwd(), "..")
  )

  for (candidate in candidates) {
    candidate_norm <- normalizePath(candidate, winslash = "/", mustWork = FALSE)
    if (dir.exists(file.path(candidate_norm, "raw")) &&
        dir.exists(file.path(candidate_norm, "script"))) {
      return(candidate_norm)
    }
  }

  stop("Could not resolve project root from working directory: ", getwd())
}

run_step <- function(base_dir, label, relative_path) {
  script_path <- file.path(base_dir, relative_path)
  message("")
  message("=== ", label, " ===")
  message("Running: ", script_path)

  tryCatch(
    {
      source(script_path, local = new.env(parent = globalenv()))
      message("Completed: ", label)
    },
    error = function(e) {
      stop(
        paste0(
          "Refresh stopped at '", label, "' (", relative_path, ").\n",
          "Error: ", conditionMessage(e)
        ),
        call. = FALSE
      )
    }
  )
}

base_dir <- resolve_base_dir()
source(file.path(base_dir, "script", "live_data_paths.txt"), local = TRUE)
live_data_dir <- get_live_data_dir(base_dir, create = TRUE)
epm_dir <- file.path(base_dir, "raw", "epm")
results_dir <- file.path(base_dir, "processed", "modelresults")
off_results_fp <- file.path(results_dir, "off_lm_1_results.csv")
def_results_fp <- file.path(results_dir, "def_lm_1_results.csv")
epm_files <- list.files(epm_dir, pattern = "^epm\\d{2}\\.csv$", full.names = TRUE)

message("End-output refresh starting from: ", base_dir)
message("This refresh will:")
message("1. pull databallr stats from 2016 through 2026")
message("2. rebuild switch history from 2016 through 2026 when doing a historical refresh")
message("3. rebuild the DPM/EPM hybrid file")
message("4. rebuild offdata.csv and defdata.csv")
message("5. rebuild the final skill output using existing model results")

if (length(epm_files) == 0) {
  stop(
    paste0(
      "No EPM files were found in ", epm_dir, ".\n",
      "This refresh cannot rebuild the combined impact file without local EPM files."
    ),
    call. = FALSE
  )
}

if (!file.exists(off_results_fp) || !file.exists(def_results_fp)) {
  stop(
    paste0(
      "Model result files are missing in ", results_dir, ".\n",
      "This refresh does not retrain models, so it requires:\n",
      off_results_fp, "\n",
      def_results_fp
    ),
    call. = FALSE
  )
}

Sys.setenv(
  DATABALLR_START_YEAR = "2016",
  DATABALLR_END_YEAR = "2026",
  SWITCH_START_YEAR = "2015",
  SWITCH_END_YEAR = "2025",
  IMPACT_EXCLUDE_YEARS = "none"
)

full_switch_history <- identical(
  tolower(trimws(Sys.getenv("REFRESH_FULL_SWITCH_HISTORY", "true"))),
  "true"
)

run_step(base_dir, "Pull Databallr Stats", file.path("script", "data", "pulldataballrstats.txt"))
if (full_switch_history) {
  run_step(base_dir, "Pull Switch History", file.path("script", "switch", "createswitchvar.txt"))
  run_step(base_dir, "Build Switch History", file.path("script", "switch", "switchprocess.txt"))
} else {
  run_step(base_dir, "Refresh Switch Latest Season", file.path("script", "switch", "switchrefresh.txt"))
}
run_step(base_dir, "Build Impact Targets", file.path("script", "data", "createimpact.txt"))
run_step(base_dir, "Build Model Datasets", file.path("script", "nbamodel", "createtestdata.txt"))
run_step(base_dir, "Build Skill Output", file.path("script", "output", "impactoutput.txt"))

message("")
message("End-output refresh completed successfully.")
message("Updated outputs:")
message(file.path(live_data_dir, "impactoutput.csv"))
message(file.path(live_data_dir, "offdata.csv"))
message(file.path(live_data_dir, "defdata.csv"))
message(file.path(live_data_dir, "skilloutput.csv"))
message(file.path(live_data_dir, "skilloutput.xlsx"))
