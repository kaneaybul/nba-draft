resolve_base_dir <- function() {
  candidates <- c(
    getwd(),
    file.path(getwd(), ".."),
    file.path(getwd(), "..", "..")
  )

  for (candidate in candidates) {
    candidate_norm <- normalizePath(candidate, winslash = "/", mustWork = FALSE)
    if (dir.exists(file.path(candidate_norm, "raw")) &&
        dir.exists(file.path(candidate_norm, "script"))) {
      return(candidate_norm)
    }
  }

  stop("Could not resolve project root from working directory: ", getwd())
}

databallr_headers <- function(method, path, query) {
  timestamp <- as.character(as.integer(Sys.time()))
  nonce <- uuid::UUIDgenerate()
  message <- paste(timestamp, nonce, toupper(method), path, query, sep = ":")
  signature <- as.character(openssl::sha256(charToRaw(message), key = charToRaw(DATABALLR_SIGNING_SECRET)))

  httr::add_headers(
    `Content-Type` = "application/json",
    `X-API-Key` = DATABALLR_API_KEY,
    `X-Timestamp` = timestamp,
    `X-Nonce` = nonce,
    `X-Signature` = signature
  )
}

read_databallr_json <- function(path, query) {
  url <- paste0(DATABALLR_API_BASE, path, "?", query)
  resp <- httr::GET(
    url,
    databallr_headers("GET", paste0("/api", path), query),
    httr::user_agent("Mozilla/5.0"),
    httr::timeout(60)
  )

  if (httr::http_error(resp)) {
    stop(
      "Databallr request failed: HTTP ", httr::status_code(resp), " ",
      httr::content(resp, as = "text", encoding = "UTF-8"),
      call. = FALSE
    )
  }

  jsonlite::fromJSON(httr::content(resp, as = "text", encoding = "UTF-8"), flatten = TRUE)
}

pull_one_year <- function(yr) {
  path <- "/supabase/player_stats_with_metrics"
  query <- paste0(
    "year=", yr,
    "&playoffs=0",
    "&min_minutes=", MIN_MINUTES,
    "&limit=", DATABALLR_LIMIT,
    "&order_by=dpm",
    "&order_direction=desc"
  )

  message("Pulling year ", yr)

  dat <- read_databallr_json(path, query)
  df <- as_tibble(dat)

  if (nrow(df) == 0) {
    message("No rows returned for ", yr)
    return(NULL)
  }

  if (!"year" %in% names(df)) {
    df <- df %>% mutate(year = yr)
  }

  if (nrow(df) >= DATABALLR_LIMIT) {
    warning(
      "Databallr returned ", nrow(df), " rows for ", yr,
      ", which reached DATABALLR_LIMIT=", DATABALLR_LIMIT,
      ". Increase DATABALLR_LIMIT or add pagination."
    )
  }

  df
}
