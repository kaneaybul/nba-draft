resolve_base_dir <- function() {
  candidates <- c(
    getwd(),
    file.path(getwd(), ".."),
    file.path(getwd(), "..", "..")
  )

  for (candidate in candidates) {
    candidate_norm <- normalizePath(candidate, winslash = "/", mustWork = FALSE)
    if (dir.exists(file.path(candidate_norm, "raw")) &&
        dir.exists(file.path(candidate_norm, "script"))) {
      return(candidate_norm)
    }
  }

  stop("Could not resolve project root from working directory: ", getwd())
}

safe_pull <- function(expr, max_tries = 3, sleep_seconds = 2) {
  last_err <- NULL

  for (i in seq_len(max_tries)) {
    out <- tryCatch(
      eval.parent(substitute(expr)),
      error = function(e) e
    )

    if (!inherits(out, "error")) {
      return(out)
    }

    last_err <- out
    message("Attempt ", i, " failed: ", out$message)

    if (i < max_tries) {
      Sys.sleep(sleep_seconds)
    }
  }

  return(NULL)
}

extract_leaguedash_table <- function(x, label = "unknown") {
  if (is.null(x)) {
    message(label, ": pull returned NULL")
    return(NULL)
  }

  if (!is.list(x)) {
    message(label, ": pull did not return a list")
    return(NULL)
  }

  if (!"LeagueDashPlayerStats" %in% names(x)) {
    message(label, ": 'LeagueDashPlayerStats' not found in response")
    return(NULL)
  }

  tab <- x[["LeagueDashPlayerStats"]]

  if (is.null(tab)) {
    message(label, ": 'LeagueDashPlayerStats' is NULL")
    return(NULL)
  }

  if (!is.data.frame(tab)) {
    message(label, ": 'LeagueDashPlayerStats' is not a data frame")
    return(NULL)
  }

  if (nrow(tab) == 0) {
    message(label, ": 'LeagueDashPlayerStats' has 0 rows")
    return(NULL)
  }

  tab
}

pull_one_season <- function(SEASON, overwrite = FALSE) {
  offense_path  <- file.path(raw_dir, paste0("offense_", SEASON, ".rds"))
  minutes_path  <- file.path(raw_dir, paste0("minutes_", SEASON, ".rds"))
  matchups_path <- file.path(raw_dir, paste0("matchups_", SEASON, ".rds"))

  if (!overwrite &&
      file.exists(offense_path) &&
      file.exists(minutes_path) &&
      file.exists(matchups_path)) {
    message("Skipping ", SEASON, " (raw files already exist).")
    return(invisible(TRUE))
  }

  message("Pulling season: ", SEASON)

  offense_raw <- safe_pull(
    nba_leaguedashplayerstats(
      season = SEASON,
      per_mode = "Per100Possessions",
      season_type = "Regular Season"
    ),
    max_tries = 3,
    sleep_seconds = 3
  )

  Sys.sleep(1)

  minutes_raw <- safe_pull(
    nba_leaguedashplayerstats(
      season = SEASON,
      season_type = "Regular Season"
    ),
    max_tries = 3,
    sleep_seconds = 3
  )

  offense <- extract_leaguedash_table(offense_raw, paste0(SEASON, " offense"))
  minutes <- extract_leaguedash_table(minutes_raw, paste0(SEASON, " minutes"))

  if (is.null(offense) || is.null(minutes)) {
    message("Skipping ", SEASON, " because offense or minutes data was invalid.")
    return(invisible(FALSE))
  }

  offense <- offense %>%
    select(
      PLAYER_ID, PLAYER_NAME, TEAM = TEAM_ABBREVIATION, AGE, GP,
      FGM, FGA, FG3M, FG3A, FTM, FTA, OREB, AST, TOV, PFD, PTS
    ) %>%
    mutate(
      PLAYER_ID = as.character(PLAYER_ID),
      across(c(AGE, GP, FGM, FGA, FG3M, FG3A, FTM, FTA, OREB, AST, TOV, PFD, PTS), as.numeric)
    )

  minutes <- minutes %>%
    select(PLAYER_ID, MIN) %>%
    mutate(
      PLAYER_ID = as.character(PLAYER_ID),
      MIN = round(as.numeric(MIN))
    )

  # Apply offensive minutes threshold before building points lookup
  offense <- offense %>%
    inner_join(minutes, by = "PLAYER_ID") %>%
    filter(MIN >= OFF_MINIMUM)

  points <- offense %>%
    select(PLAYER_ID, PTS)

  # Eligible defenders for this season
  defender_minutes <- minutes %>%
    filter(MIN >= DEF_MINIMUM) %>%
    rename(DEF_PLAYER_ID = PLAYER_ID,
           DEF_MIN = MIN)

  matchup_list <- vector("list", length(list_IDs))

  for (i in seq_along(list_IDs)) {
    id <- list_IDs[i]

    slug <- team_info %>%
      filter(idTeam == as.numeric(id)) %>%
      pull(slugTeam)

    matchup_data <- tryCatch({
      Sys.sleep(0.5)
      safe_pull(
        nba_leagueseasonmatchups(def_team_id = id, season = SEASON)[[1]],
        max_tries = 2,
        sleep_seconds = 1
      )
    }, error = function(e) {
      message("No matchup data for team ID ", id, " in ", SEASON, ": ", e$message)
      return(NULL)
    })

    if (!is.null(matchup_data) && is.data.frame(matchup_data) && nrow(matchup_data) > 0) {
      matchup_list[[i]] <- matchup_data %>%
        mutate(
          OFF_PLAYER_ID = as.character(OFF_PLAYER_ID),
          DEF_PLAYER_ID = as.character(DEF_PLAYER_ID)
        ) %>%
        # keep only offensive players who cleared minute threshold
        inner_join(points, by = c("OFF_PLAYER_ID" = "PLAYER_ID")) %>%
        # keep only defenders who cleared minute threshold
        inner_join(defender_minutes, by = "DEF_PLAYER_ID") %>%
        mutate(
          slugTeam = slug,
          season = SEASON
        )
    }

    message("Processed ", i, " / ", length(list_IDs), " teams for ", SEASON)
  }

  matchups <- bind_rows(matchup_list)

  if (nrow(matchups) == 0) {
    message("Skipping save for ", SEASON, " because matchup data was empty after filters.")
    return(invisible(FALSE))
  }

  saveRDS(offense, offense_path)
  saveRDS(minutes, minutes_path)
  saveRDS(matchups, matchups_path)

  message("Saved raw files for ", SEASON)
  invisible(TRUE)
}

results <- map_lgl(season_vec, ~ pull_one_season(.x, overwrite = OVERWRITE))
